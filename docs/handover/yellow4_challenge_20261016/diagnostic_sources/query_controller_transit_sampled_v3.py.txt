"""Query-only provisional air path IK/FK; no device motion clients."""
import json,sys
from pathlib import Path
import numpy as np
from scipy.spatial.transform import Rotation, Slerp
import rclpy
from dsr_msgs2.srv import Ikin,Fkin,GetSolutionSpace
HERE=Path(__file__).resolve().parent
ROOT=HERE.parents[1]
sys.path.insert(0,str(ROOT/'robot_control_integration'))
from app.edge_first_seating import pose_matrix,matrix_pose
rclpy.init();node=rclpy.create_node('yellow4_ik_query_only')
clients={name:node.create_client(kind,'/dsr01/dsr_controller2/'+path) for name,kind,path in [('sol',GetSolutionSpace,'aux_control/get_solution_space'),('ik',Ikin,'motion/ikin'),('fk',Fkin,'motion/fkin')]}
def query(name,kind,**values):
    c=clients[name]
    if not c.wait_for_service(timeout_sec=4):raise RuntimeError(name+'_SERVICE_UNAVAILABLE')
    req=kind.Request()
    for k,v in values.items():setattr(req,k,v)
    future=c.call_async(req);rclpy.spin_until_future_complete(node,future,timeout_sec=4)
    if not future.done():raise RuntimeError(name+'_TIMEOUT')
    response=future.result()
    if response is None or not response.success:raise RuntimeError(name+'_FAILED')
    return response
try:
    capture=json.loads((ROOT/'logs/yellow4-field-seated-grasp/capture-1791965183566613886/capture.json').read_text())
    sol=query('sol',GetSolutionSpace,pos=capture['sample']['posj']).sol_space
    path=json.loads((HERE/'transit-candidate-v3.preview-only.json').read_text())
    start=json.loads((ROOT/'logs/yellow4-field-joint-review/capture-1791967444118846356/capture.json').read_text())['sample']['posx']
    sampled=[]
    for endpoint in path['waypoints']:
        end=endpoint['tcp_mm_zyz_deg'];A=pose_matrix(start);B=pose_matrix(end)
        rotations=Rotation.from_matrix(np.stack([A[:3,:3],B[:3,:3]]));angle=np.degrees((rotations[0].inv()*rotations[1]).magnitude())
        count=max(1,int(np.ceil(np.linalg.norm(B[:3,3]-A[:3,3])/20)),int(np.ceil(angle/5)))
        interpolation=Slerp([0,1],rotations)
        for step in range(1,count+1):
            t=step/count;T=np.eye(4);T[:3,:3]=interpolation(t).as_matrix();T[:3,3]=(1-t)*A[:3,3]+t*B[:3,3]
            sampled.append({'label':endpoint['label']+'/'+str(step)+'of'+str(count),'tcp_mm_zyz_deg':matrix_pose(T)})
        start=end
    path['waypoints']=sampled
    result={'status':'PROVISIONAL_GEOMETRY_IK_DIAGNOSTIC_NOT_EXECUTABLE','physical_commands':0,'execution_allowed':False,'assumed_base_z_translation_mm':0,'sol_space':sol,'waypoints':[],'note':'5deg and contact axis/grasp geometry are assumptions. IK/FK does not validate clearance, physical grasp or engagement.'}
    previous=None
    for w in path['waypoints']:
        target=list(w['tcp_mm_zyz_deg'])
        joints=[float(v) for v in query('ik',Ikin,pos=target,sol_space=sol,ref=0).conv_posj]
        actual=[float(v) for v in query('fk',Fkin,pos=joints,ref=0).conv_posx]
        A=pose_matrix(target);B=pose_matrix(actual)
        angle=float(np.degrees(np.arccos(np.clip((np.trace(A[:3,:3].T@B[:3,:3])-1)/2,-1,1))))
        result['waypoints'].append({'label':w['label'],'target_tcp':target,'joints_deg':joints,'fk_tcp':actual,'position_error_mm':float(np.linalg.norm(A[:3,3]-B[:3,3])),'so3_error_deg':angle,'delta_joints_from_previous_deg':None if previous is None else [a-b for a,b in zip(joints,previous)]})
        previous=joints
    result['invalid_endpoint_labels']=[w['label'] for w in result['waypoints'] if w['position_error_mm']>1 or w['so3_error_deg']>0.1 or any(not np.isfinite(q) or abs(q)>360 for q in w['joints_deg'])]
    result['large_adjacent_joint_changes']=[{'label':w['label'],'delta':w['delta_joints_from_previous_deg']} for w in result['waypoints'] if w['delta_joints_from_previous_deg'] is not None and max(abs(v) for v in w['delta_joints_from_previous_deg'])>10]
    result['status']='REJECTED_IK_ENDPOINTS' if result['invalid_endpoint_labels'] else 'SAMPLED_IK_REVIEW_REQUIRED' if result['large_adjacent_joint_changes'] else 'SAMPLED_IK_CONSISTENT_COLLISION_STILL_UNVERIFIED'
    out=HERE/'controller-ik-transit-sampled-v3.json';out.write_text(json.dumps(result,indent=2)+'\n');print('IK_QUERY_SAVED:',out,'physical_commands=0; provisional only')
finally:node.destroy_node();rclpy.shutdown()
